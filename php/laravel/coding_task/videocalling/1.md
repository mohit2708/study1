```php
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Twilio Video Call</title>
    <meta name="viewport" content="width=device-width, initial-scale=1">

    <!-- Twilio Video SDK -->
    <script src="https://sdk.twilio.com/js/video/releases/2.28.1/twilio-video.min.js"></script>

    <!-- Axios -->
    <script src="https://cdn.jsdelivr.net/npm/axios/dist/axios.min.js"></script>

    <style>
        body {
            font-family: Arial, sans-serif;
            padding: 20px;
        }

        button {
            margin: 5px;
            padding: 10px 16px;
            font-size: 14px;
            cursor: pointer;
        }

        #video-chat-window {
            display: flex;
            flex-wrap: wrap;
            margin-top: 20px;
        }

        video {
            width: 300px;
            margin: 10px;
            border-radius: 8px;
            background: black;
        }
    </style>
</head>
<body>

<h2>Twilio Video Calling</h2>

<button id="openCameraBtn">📷 Open Camera</button>
<button id="startCallBtn" disabled>▶ Start Call</button>
<button id="endCallBtn" disabled>⛔ End Call</button>

<div id="video-chat-window"></div>

<script>
    let room = null;
    let accessToken = null;
    let roomName = null;
    let localTracks = [];

    const userId = {{ auth()->id() }};

    /* -------------------------------
       1️⃣ Get Access Token
    -------------------------------- */
    function getAccessToken() {
        return axios.get(`/api/twilio_access_token/${userId}`)
            .then(res => {
                accessToken = res.data.token;
                roomName = res.data.roomName;

                if (typeof accessToken !== 'string') {
                    throw new Error('Token is not string');
                }
            });
    }

    /* -------------------------------
       2️⃣ Open Camera (Preview Only)
    -------------------------------- */
    document.getElementById('openCameraBtn').onclick = async () => {

        if (localTracks.length > 0) return;

        await getAccessToken();

        localTracks = await Twilio.Video.createLocalTracks({
            audio: true,
            video: { width: 640, height: 480 }
        });

        const container = document.getElementById('video-chat-window');

        localTracks.forEach(track => {
            container.appendChild(track.attach());
        });

        document.getElementById('startCallBtn').disabled = false;
        document.getElementById('endCallBtn').disabled = false;
    };

    /* -------------------------------
       3️⃣ Start Call
    -------------------------------- */
    document.getElementById('startCallBtn').onclick = () => {

        if (!accessToken || localTracks.length === 0) {
            alert('Camera not opened');
            return;
        }

        Twilio.Video.connect(accessToken, {
            name: roomName,
            tracks: localTracks
        }).then(joinedRoom => {

            room = joinedRoom;
            console.log('Connected to room:', room.name);

            room.participants.forEach(attachParticipantTracks);

            room.on('participantConnected', attachParticipantTracks);
            room.on('participantDisconnected', detachParticipantTracks);

        }).catch(err => {
            console.error('Connection failed:', err);
        });
    };

    /* -------------------------------
       4️⃣ End Call
    -------------------------------- */
    document.getElementById('endCallBtn').onclick = () => {

        if (room) {
            room.disconnect();
            room = null;
        }

        localTracks.forEach(track => {
            track.stop();
            track.detach().forEach(el => el.remove());
        });

        localTracks = [];

        document.getElementById('startCallBtn').disabled = true;
        document.getElementById('endCallBtn').disabled = true;
    };

    /* -------------------------------
       5️⃣ Remote Participants
    -------------------------------- */
    function attachParticipantTracks(participant) {
        const container = document.getElementById('video-chat-window');

        participant.tracks.forEach(publication => {
            if (publication.isSubscribed) {
                container.appendChild(publication.track.attach());
            }
        });

        participant.on('trackSubscribed', track => {
            container.appendChild(track.attach());
        });

        participant.on('trackUnsubscribed', track => {
            track.detach().forEach(el => el.remove());
        });
    }

    function detachParticipantTracks(participant) {
        participant.tracks.forEach(publication => {
            if (publication.track) {
                publication.track.detach().forEach(el => el.remove());
            }
        });
    }

    /* -------------------------------
       6️⃣ Cleanup on page close
    -------------------------------- */
    window.addEventListener('beforeunload', () => {
        if (room) room.disconnect();
    });
</script>

</body>
</html>

```
