### Query parameter
* A query parameter is a key-value pair included in the URL of a web request to pass additional, optional data to the server. They are most commonly used with GET requests to perform actions like filtering, sorting, or paginating data.
* A query string begins with a question mark (?) and separates multiple parameters with an ampersand (&).
* For example, in the URL http://example.com/items?category=books&sort=asc, the query parameters are:
```python
from fastapi import FastAPI, Query

app = FastAPI()

fake_items_db = [{"item_name": "Foo"}, {"item_name": "Bar"}, {"item_name": "Baz"}]


@app.get("/items/")
async def read_item(skip: int = 0, limit: int = 10):
    return fake_items_db[skip : skip + limit]

@app.get("/search/")
async def search(q: str = Query(None, min_length=3, max_length=50)):
    return {"message": f"Searching for: {q}"}
```

### Optional parameters
* Types of Optional parameters
  * Optional
  * Union operator (|) for Python 3.10+
  * typing.Union for Python 3.9 and earlier

```python
# here q is optional
from fastapi import FastAPI

app = FastAPI()

@app.get("/items/{item_id}")
async def read_item(item_id: str, q: str | None = None):
    if q:
        return {"item_id": item_id, "q": q}
    return {"item_id": item_id}
```
* Using Optional Method
```python
from fastapi import FastAPI
from typing import Optional

app = FastAPI()

@app.get("/items/{item_id}")
async def read_item(item_id: str, q: Optional[int] = None):
    if q:
        return {"item_id": item_id, "q": q}
    return {"item_id": item_id}
```
* Union operator (|) for Python 3.10+
```python
from fastapi import FastAPI
# No need to import `Optional` or `Union`

app = FastAPI()

@app.get("/items/")
async def optional_query_parameter(id: int | None = None):
    if id:
        return {"get return id": id}
    return {"message": "ID is optional. No ID was provided."}
```
* typing.Union for Python 3.9 and earlier
```python
from fastapi import FastAPI
from typing import Union

app = FastAPI()

@app.get("/items/")
async def optional_query_parameter(id: Union[int, None] = None):
    if id:
        return {"get return id": id}
    return {"message": "ID is optional. No ID was provided."}
```

### Query parameter type conversion
```python
from fastapi import FastAPI

app = FastAPI()

@app.get("/items/{item_id}")
async def read_item(item_id: str, q: str | None = None, short: bool = False):
    item = {"item_id": item_id}
    if q:
        item.update({"q": q})
    if not short:
        item.update(
            {"description": "This is an amazing item that has a long description"}
        )
    return item
```

### Required query parameters
```python
from fastapi import FastAPI

app = FastAPI()

@app.get("/items/{item_id}")
async def read_user_item(item_id: str, needy: str):
    item = {"item_id": item_id, "needy": needy}
    return item
```
* we can define some parameters as required, some as having a default value, and some entirely optional.
```python
from fastapi import FastAPI

app = FastAPI()

@app.get("/items/{item_id}")
async def read_user_item(
    item_id: str, needy: str, skip: int = 0, limit: int | None = None
):
    item = {"item_id": item_id, "needy": needy, "skip": skip, "limit": limit}
    return item
```

#### Query Parameters and String Validations
```python
from fastapi import FastAPI

app = FastAPI()

@app.get("/items/")
async def read_items(q: str | None = None):
    results = {"items": [{"item_id": "Foo"}, {"item_id": "Bar"}]}
    if q:
        results.update({"q": q})
    return results
```

#### Additional validation
* Import Query and Annotated
```python
from typing import Annotated
from fastapi import FastAPI, Query

app = FastAPI()

@app.get("/items/")
async def read_items(q: Annotated[str | None, Query(min_length=3, max_length=50)] = None):
    results = {"items": [{"item_id": "Foo"}, {"item_id": "Bar"}]}
    if q:
        results.update({"q": q})
    return results
```

#### Add regular expressions
```python
from typing import Annotated
from fastapi import FastAPI, Query

app = FastAPI()

@app.get("/items/")
async def read_items(
    q: Annotated[
        str | None, Query(min_length=3, max_length=50, pattern="^fixedquery$")
    ] = None,
):
    results = {"items": [{"item_id": "Foo"}, {"item_id": "Bar"}]}
    if q:
        results.update({"q": q})
    return results
```

#### Query parameter list / multiple values
```python
from typing import Annotated
from fastapi import FastAPI, Query

app = FastAPI()

@app.get("/items/")
async def read_items(q: Annotated[list[str] | None, Query()] = None):
    query_items = {"q": q}
    return query_items
```

#### Declare metadata
* like alias
* like title
```python
from typing import Annotated
from fastapi import FastAPI, Path, Query

app = FastAPI()

@app.get("/items/{item_id}")
async def read_items(
    item_id: Annotated[int, Path(title="The ID of the item to get", alias="item-dquery")],
    q: Annotated[str | None, Query(alias="item-query")] = None,
):
    results = {"item_id": item_id}
    if q:
        results.update({"q": q})
    return results
```

#### Number validations: greater than and less than or equal
```python
from typing import Annotated

from fastapi import FastAPI, Path

app = FastAPI()


@app.get("/items/{item_id}")
async def read_items(
    item_id: Annotated[int, Path(title="The ID of the item to get", gt=1, le=1000)],
    q: str,
):
    results = {"item_id": item_id}
    if q:
        results.update({"q": q})
    return results
```
